1"""Detecta os test smells do projeto usando o motor de detecção do TEMPY, 

2sem a interface gráfica, e salva os resultados em um formato padronizado. 

3 

4Uso (sempre a partir da pasta raiz do projeto): 

5 

6 python extract_smells_tempy.py --etapa before # antes da refatoração (salva) 

7 python extract_smells_tempy.py --etapa after # depois da refatoração (salva) 

8 python extract_smells_tempy.py # conferência rápida (não salva nada) 

9 python extract_smells_tempy.py --arquivo tests/test_x.py # conferência de um arquivo (não salva) 

10"""

34def load_tempy(tempy_dir): 

35 """Importa o parser do TEMPY a partir da pasta clonada.""" 

36 assets = os.path.join(os.path.abspath(tempy_dir), "assets") 

37 if not os.path.isfile(os.path.join(assets, "python_parser.py")): 

38 sys.exit(f"[erro] TEMPY não encontrado em: {os.path.abspath(tempy_dir)}\n" 

39 f" Ajuste TEMPY_DIR no início do script ou use --tempy <caminho>.") 

40 sys.path.insert(0, assets) 

41 import python_parser 

42 # O TEMPY abre os arquivos sem definir a codificação, o que quebra no Windows 

43 # com arquivos UTF-8 acentuados. Forçamos UTF-8 apenas dentro do parser. 

44 python_parser.open = functools.partial(open, encoding="utf-8", errors="ignore") 

45 return python_parser.PythonParser

48def is_ignored_dir(name): 

49 return name.startswith(".") or name.startswith("metrics-") or name in IGNORED_DIRS

52def is_test_file(path): 

53 """Mesmo critério do TEMPY: o arquivo importa unittest/pytest ou tem linha começando com assert.""" 

54 with open(path, encoding="utf-8", errors="ignore") as f: 

55 for line in f: 

56 if line.find("assert") == 0: 

57 return True 

58 if any(t in line for t in ("import unittest", "import pytest", "from unittest", "from pytest")): 

59 return True 

60 return False

63def find_test_files(root): 

64 files = [] 

65 for dirpath, dirnames, filenames in os.walk(root): 

66 dirnames[:] = sorted(d for d in dirnames if not is_ignored_dir(d)) 

67 for name in sorted(filenames): 

68 path = os.path.join(dirpath, name) 

69 if name.endswith(".py") and name != SCRIPT_NAME and is_test_file(path): 

70 files.append(path) 

71 return files

74def analyze(parser_cls, path): 

75 """Executa a detecção do TEMPY em um arquivo. Retorna (ocorrências, erro).""" 

76 try: 

77 with contextlib.redirect_stdout(io.StringIO()): # o TEMPY imprime muita coisa 

78 parser = parser_cls(path) 

79 occurrences = parser.start() if parser.ast_parser else parser.start2() 

80 return occurrences, None 

81 except Exception as e: # arquivo que o TEMPY não conseguiu analisar 

82 return [], f"{type(e).__name__}: {e}"

85def git_commit(path): 

86 try: 

87 r = subprocess.run(["git", "rev-parse", "HEAD"], cwd=path, capture_output=True, text=True) 

88 return r.stdout.strip() or None 

89 except OSError: 

90 return None

93def run(parser_cls, root, files): 

94 rows, errors, seen = [], [], set() 

95 for path in files: 

96 rel = os.path.relpath(path, root).replace(os.sep, "/") 

97 occurrences, error = analyze(parser_cls, path) 

98 if error: 

99 errors.append({"arquivo": rel, "erro": error}) 

100 for occ in occurrences: 

101 lines = ";".join(str(n) for n in occ.lines) 

102 key = (rel, occ.test_smell_type, occ.method_name, lines) 

103 if key in seen: # descarta ocorrências idênticas repetidas 

104 continue 

105 seen.add(key) 

106 rows.append({"arquivo": rel, "smell": occ.test_smell_type, 

107 "metodo": occ.method_name or "(nível do módulo)", "linhas": lines}) 

108 return rows, errors

111def print_summary(rows, n_files, errors): 

112 print(f"\n{len(rows)} test smells encontrados em {n_files} arquivos de teste analisados.\n") 

113 for smell, n in Counter(r["smell"] for r in rows).most_common(): 

114 print(f" {n:5d} {smell}") 

115 if errors: 

116 print(f"\n[aviso] {len(errors)} arquivo(s) não puderam ser analisados pelo TEMPY.")

119def write_html(path, summary, rows): 

120 by_type = "".join(f"<tr><td>{html.escape(s)}</td><td>{n}</td></tr>" 

121 for s, n in summary["por_tipo"].items()) 

122 body = "".join( 

123 f"<tr><td>{i}</td><td>{html.escape(r['arquivo'])}</td><td>{html.escape(r['smell'])}</td>" 

124 f"<td>{html.escape(r['metodo'])}</td><td>{html.escape(r['linhas'].replace(';', ', '))}</td></tr>" 

125 for i, r in enumerate(rows, 1)) 

126 page = f"""<!DOCTYPE html> 

127<html lang="pt-br"><head><meta charset="utf-8"><title>Test smells ({summary['etapa']})</title> 

128<style> 

129body {{ font-family: system-ui, sans-serif; margin: 2rem; }} 

130table {{ border-collapse: collapse; margin-bottom: 2rem; }} 

131th, td {{ border: 1px solid #999; padding: 4px 8px; text-align: left; }} 

132th {{ background: #eee; }} 

133</style></head><body> 

134<h1>Test smells detectados pelo TEMPY ({summary['etapa']})</h1> 

135<p>Data: {summary['data']}<br>Commit do projeto: {summary['commit_projeto']}<br> 

136Arquivos de teste analisados: {summary['arquivos_analisados']}<br> 

137Total de test smells: <strong>{summary['total_smells']}</strong></p> 

138<h2>Por tipo</h2><table><tr><th>Test smell</th><th>Ocorrências</th></tr>{by_type}</table> 

139<h2>Ocorrências</h2> 

140<table><tr><th>#</th><th>Arquivo</th><th>Test smell</th><th>Método</th><th>Linhas</th></tr>{body}</table> 

141</body></html>""" 

142 with open(path, "w", encoding="utf-8") as f: 

143 f.write(page)

146def main(): 

147 p = argparse.ArgumentParser(description="Detecta test smells com o TEMPY.") 

148 p.add_argument("--etapa", choices=("before", "after"), 

149 help="salva os resultados em metrics-<etapa>-tempy/. Sem esta opção, nada é salvo.") 

150 p.add_argument("--arquivo", help="analisa apenas este arquivo de teste (conferência; não salva)") 

151 p.add_argument("--tempy", default=TEMPY_DIR, help="pasta onde o TEMPY foi clonado") 

152 args = p.parse_args() 

153 

154 if args.etapa and args.arquivo: 

155 p.error("--arquivo serve só para conferência e não pode ser usado com --etapa") 

156 

157 root = os.getcwd() 

158 parser_cls = load_tempy(args.tempy) 

159 

160 if args.arquivo: 

161 if not os.path.isfile(args.arquivo): 

162 sys.exit(f"[erro] arquivo não encontrado: {args.arquivo}") 

163 rows, errors = run(parser_cls, root, [args.arquivo]) 

164 for r in rows: 

165 print(f" {r['smell']:<30} {r['metodo']:<40} linhas {r['linhas'].replace(';', ', ')}") 

166 print_summary(rows, 1, errors) 

167 for e in errors: 

168 print(f" {e['arquivo']}: {e['erro']}") 

169 return 

170 

171 out_dir = f"metrics-{args.etapa}-tempy" if args.etapa else None 

172 if out_dir and os.path.exists(out_dir): 

173 sys.exit(f"[erro] a pasta {out_dir}/ já existe. Os resultados não serão sobrescritos.\n" 

174 f" Se precisar refazer a medição, fale com o monitor antes de apagar a pasta.") 

175 

176 files = find_test_files(root) 

177 if not files: 

178 sys.exit("[erro] nenhum arquivo de teste encontrado. Execute o script na pasta raiz do projeto.") 

179 rows, errors = run(parser_cls, root, files) 

180 print_summary(rows, len(files), errors) 

181 

182 if not out_dir: 

183 print("\n(modo conferência: nenhum arquivo foi salvo)") 

184 return 

185 

186 os.makedirs(out_dir) 

187 summary = { 

188 "etapa": args.etapa, 

189 "data": datetime.now().isoformat(timespec="seconds"), 

190 "commit_projeto": git_commit(root), 

191 "commit_tempy": git_commit(os.path.abspath(args.tempy)), 

192 "arquivos_analisados": len(files), 

193 "arquivos_com_erro": len(errors), 

194 "total_smells": len(rows), 

195 "por_tipo": dict(Counter(r["smell"] for r in rows).most_common()), 

196 "por_arquivo": dict(Counter(r["arquivo"] for r in rows).most_common()), 

197 "erros": errors, 

198 } 

199 with open(os.path.join(out_dir, "smells.csv"), "w", newline="", encoding="utf-8") as f: 

200 writer = csv.DictWriter(f, fieldnames=["arquivo", "smell", "metodo", "linhas"]) 

201 writer.writeheader() 

202 writer.writerows(rows) 

203 with open(os.path.join(out_dir, "resumo.json"), "w", encoding="utf-8") as f: 

204 json.dump(summary, f, ensure_ascii=False, indent=2) 

205 write_html(os.path.join(out_dir, "relatorio.html"), summary, rows) 

206 print(f"\nResultados salvos em {out_dir}/ (smells.csv, resumo.json, relatorio.html)")

208 

209if __name__ == "__main__": 

210 main()
